package wips.powers.mixin;

import com.awakenedmod.awakened.client.render.VfxRenderer;
import com.awakenedmod.awakened.content.vfx.VfxEntity;
import java.util.EnumSet;
import java.util.Set;
import net.minecraft.client.renderer.culling.Frustum;
import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfoReturnable;

/**
 * REEMPLAZO: las formas decorativas de Awakened (esferas, anillos, domos, halos, salpicaduras,
 * sonic cracks, aros cineticos, grietas de Ruin, charcos/anillos de agua...) ya no se dibujan.
 * Las formas que SON el ataque (rayo, arco, chorros y olas de agua, geiser, portales, bolt) siguen.
 */
@Mixin(value = VfxRenderer.class, remap = false)
public abstract class VfxRendererReplaceMixin {
    private static final Set<VfxEntity.Kind> HIDDEN = hidden();

    private static Set<VfxEntity.Kind> hidden() {
        Set<VfxEntity.Kind> s = EnumSet.noneOf(VfxEntity.Kind.class);
        for (VfxEntity.Kind k : VfxEntity.Kind.values()) {
            String n = k.name();
            boolean keep = n.equals("BEAM") || n.equals("ARC") || n.equals("PORTAL") || n.equals("PILLAR")
                    || n.equals("VELOCITY_BOLT") || n.equals("VELOCITY_WORMHOLE") || n.equals("VELOCITY_TUNNEL_MAW")
                    || n.equals("WATER_STREAM") || n.equals("WATER_GEYSER") || n.equals("WATER_TENDRIL") || n.equals("WATER_WAVE")
                    || n.equals("TORNADO") || n.equals("AIR_WALL") || n.startsWith("TIME_") || n.startsWith("STASIS") || n.equals("REWIND");
            if (!keep) {
                s.add(k);
            }
        }
        return s;
    }

    @Inject(method = "shouldRender(Lcom/awakenedmod/awakened/content/vfx/VfxEntity;Lnet/minecraft/client/renderer/culling/Frustum;DDD)Z",
            at = @At("HEAD"), cancellable = true)
    private void wips$hide(VfxEntity entity, Frustum frustum, double x, double y, double z, CallbackInfoReturnable<Boolean> cir) {
        if (HIDDEN.contains(entity.kind())) {
            cir.setReturnValue(false);
        }
    }
}
